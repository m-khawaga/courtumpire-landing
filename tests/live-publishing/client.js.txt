let busy=false;const out=document.querySelector('#result'),button=document.querySelector('#next');
button.addEventListener('click',async()=>{if(busy)return;busy=true;button.disabled=true;
try{const status=await fetch('/publishing-test/status').then(r=>r.json());if(status.runs.some(r=>!r.passed))throw Error('A previous operation failed; stop.');const step=status.runs.length;if(step>23){out.textContent='All isolated operations completed. CPU review still required.';return;}
const session=await fetch('/publishing-test/session').then(r=>r.json());out.textContent='Running step '+step+'…';const start=performance.now();const response=await fetch('/publishing-test/step',{method:'POST',headers:{'content-type':'application/json','x-csrf-token':session.csrf},body:JSON.stringify({step})});const result=await response.json();result.wallMs=Math.round(performance.now()-start);out.textContent=JSON.stringify(result,null,2);if(!response.ok||!result.passed)throw Error('Gate failed. Stop.');button.textContent='Run next isolated operation (after CPU review)';button.disabled=false;
}catch(e){out.textContent+='\n'+e.message;}finally{busy=false;}});
